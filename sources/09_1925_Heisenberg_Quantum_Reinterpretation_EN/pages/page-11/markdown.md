$$a _ {1} ^ {2} (n) \quad \text { or } \quad a ^ {2} (n, n - 1) = \frac {n + \operatorname{const} h}{\pi m \omega_ {0}} \tag {46}$$

$$a ^ {2} (n, n - 1) = \frac {n h}{\pi m \omega_ {0}} \tag {47}$$
